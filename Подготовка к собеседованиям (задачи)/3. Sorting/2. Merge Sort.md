class Solution:
    def merge_sort(self, nums: list[int]) -> list[int]:
        if len(nums) <=1:
            return nums

        mid = len(nums) // 2
        left_nums = self.merge_sort(nums[:mid])
        right_nums = self.merge_sort(nums[mid:])

        return self.merge(left_nums, right_nums)

    @staticmethod
    def merge(left_nums: list[int], right_nums: list[int]):
        l = r = 0
        result = []
        while l < len(left_nums) and r < len(right_nums):
            if left_nums[l] <= right_nums[r]:
                result.append(left_nums[l])
                l += 1
            else:
                result.append(right_nums[r])
                r += 1

        result.extend(left_nums[l:])
        result.extend(right_nums[r:])

        return result

sol = Solution()
print(sol.merge_sort([-10, 9, 5, 44, 7, 2, 8, 1, 4, 0, 3, -44, 6]))
