class Solution:
    def quick_sort(self, nums: list[int]) -> list[int]:
        if len(nums) < 2:
            return nums
        pivot = nums[len(nums) // 2]

        left_nums = [x for x in nums if x < pivot]
        mid_nums = [x for x in nums if x == pivot]
        right_nums = [x for x in nums if x > pivot]

        return self.quick_sort(left_nums) + mid_nums + self.quick_sort(right_nums)

sol = Solution()
print(sol.quick_sort([9, 5, 44, 7, 2, 8, 1, 4, 0, 3, 6]))
